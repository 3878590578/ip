// ============================================================
// Cloudflare Worker
// GitHub ProxyIP → SG-AS16509 → VLESS
//
// /cf      订阅
// /update  手动更新
// /info    查看状态
//
// Cron：每天北京时间 12:00
// UTC = 04:00
// ============================================================


// ============================================================
// 配置
// ============================================================

const SOURCE_URL =
  "https://raw.githubusercontent.com/3878590578/ip/refs/heads/main/新加坡16509.txt";

const UUID =
  "19db41ac-5f66-4a46-a733-a03c0e0ef04b";

const DOMAIN =
  "sg16.mofaxueyuan.kdns.fr";

const KV_KEY =
  "SG_AS16509_SUBSCRIPTION";


// ============================================================
// Cloudflare Worker
// ============================================================

export default {

  async fetch(request, env) {

    const url =
      new URL(request.url);

    const path =
      url.pathname;


    // ========================================================
    // /cf
    // ========================================================

    if (path === "/cf") {

      const result =
        await env.SUB_KV.get(KV_KEY);

      if (!result) {

        return new Response(
          "订阅暂未生成，请访问 /update 更新",
          {
            status: 404,
            headers: {
              "Content-Type":
                "text/plain; charset=utf-8"
            }
          }
        );
      }


      return new Response(
        result,
        {
          headers: {
            "Content-Type":
              "text/plain; charset=utf-8",

            "Cache-Control":
              "no-cache, no-store, must-revalidate"
          }
        }
      );
    }


    // ========================================================
    // /update
    // ========================================================

    if (path === "/update") {

      try {

        const result =
          await processSource();


        if (!result.nodes.length) {

          return new Response(
            JSON.stringify(
              {
                success: false,
                message:
                  "没有找到 SG-AS16509 节点"
              },
              null,
              2
            ),
            {
              status: 500,
              headers: {
                "Content-Type":
                  "application/json; charset=utf-8"
              }
            }
          );
        }


        // 保存订阅
        await env.SUB_KV.put(
          KV_KEY,
          result.output
        );


        // 保存更新时间
        const updated =
          new Date().toISOString();

        await env.SUB_KV.put(
          "SG_UPDATED_TIME",
          updated
        );


        // 保存节点数量
        await env.SUB_KV.put(
          "SG_NODE_COUNT",
          String(result.nodes.length)
        );


        return new Response(
          JSON.stringify(
            {
              success: true,
              count:
                result.nodes.length,
              updated:
                updated
            },
            null,
            2
          ),
          {
            headers: {
              "Content-Type":
                "application/json; charset=utf-8"
            }
          }
        );

      } catch (error) {

        return new Response(
          JSON.stringify(
            {
              success: false,
              error:
                String(error)
            },
            null,
            2
          ),
          {
            status: 500,
            headers: {
              "Content-Type":
                "application/json; charset=utf-8"
            }
          }
        );
      }
    }


    // ========================================================
    // /info
    // ========================================================

    if (path === "/info") {

      const updated =
        await env.SUB_KV.get(
          "SG_UPDATED_TIME"
        );

      const count =
        await env.SUB_KV.get(
          "SG_NODE_COUNT"
        );


      return new Response(
        JSON.stringify(
          {
            country:
              "SG",

            as:
              "AS16509",

            count:
              Number(count || 0),

            updated:
              updated || null,

            subscription:
              new URL(
                "/cf",
                url.origin
              ).toString()
          },
          null,
          2
        ),
        {
          headers: {
            "Content-Type":
              "application/json; charset=utf-8"
          }
        }
      );
    }


    // ========================================================
    // 首页
    // ========================================================

    if (path === "/") {

      return new Response(
        "SG-AS16509 VLESS Worker\n\n" +
        "/cf      订阅\n" +
        "/update  手动更新\n" +
        "/info    查看状态",
        {
          headers: {
            "Content-Type":
              "text/plain; charset=utf-8"
          }
        }
      );
    }


    // ========================================================
    // 404
    // ========================================================

    return new Response(
      "Not Found",
      {
        status: 404
      }
    );
  },


  // ==========================================================
  // Cron
  // 每天 UTC 04:00
  // = 北京时间 12:00
  // ==========================================================

  async scheduled(event, env, ctx) {

    ctx.waitUntil(
      updateSubscription(env)
    );
  }
};


// ============================================================
// 自动更新
// ============================================================

async function updateSubscription(env) {

  try {

    const result =
      await processSource();


    if (!result.nodes.length) {

      console.log(
        "更新失败：没有找到 SG-AS16509 节点"
      );

      return;
    }


    await env.SUB_KV.put(
      KV_KEY,
      result.output
    );


    const updated =
      new Date().toISOString();


    await env.SUB_KV.put(
      "SG_UPDATED_TIME",
      updated
    );


    await env.SUB_KV.put(
      "SG_NODE_COUNT",
      String(result.nodes.length)
    );


    console.log(
      "更新成功：" +
      result.nodes.length +
      " 个 SG-AS16509 节点"
    );

  } catch (error) {

    console.log(
      "更新失败：",
      String(error)
    );
  }
}


// ============================================================
// 处理 GitHub 数据
// ============================================================

async function processSource() {

  // ----------------------------------------------------------
  // 1. 下载 GitHub
  // ----------------------------------------------------------

  const response =
    await fetch(
      SOURCE_URL,
      {
        headers: {
          "User-Agent":
            "Cloudflare-Worker"
        }
      }
    );


  if (!response.ok) {

    throw new Error(
      "GitHub 下载失败：" +
      response.status
    );
  }


  const text =
    await response.text();


  if (!text.trim()) {

    throw new Error(
      "GitHub 文件为空"
    );
  }


  // ----------------------------------------------------------
  // 2. 分行
  // ----------------------------------------------------------

  const lines =
    text.split(/\r?\n/);


  const items = [];


  // ----------------------------------------------------------
  // 3. 解析
  // ----------------------------------------------------------

  for (const line of lines) {

    const parsed =
      parseLine(line);


    if (!parsed) {
      continue;
    }


    // --------------------------------------------------------
    // 判断新加坡
    // --------------------------------------------------------

    const country =
      detectCountry(
        parsed.name
      );


    if (country !== "SG") {
      continue;
    }


    // --------------------------------------------------------
    // 判断 AS16509
    // --------------------------------------------------------

    const as =
      detectAS(
        parsed.name
      );


    if (as !== "AS16509") {
      continue;
    }


    items.push(
      {
        ip:
          parsed.ip,

        port:
          parsed.port,

        name:
          parsed.name,

        country:
          country,

        as:
          as
      }
    );
  }


  // ----------------------------------------------------------
  // 4. IP 数字排序
  // ----------------------------------------------------------

  items.sort(
    function(a, b) {

      return (
        ipToNumber(a.ip) -
        ipToNumber(b.ip)
      );
    }
  );


  // ----------------------------------------------------------
  // 5. IP:端口去重
  // ----------------------------------------------------------

  const seen =
    new Set();

  const uniqueItems =
    [];


  for (const item of items) {

    const key =
      item.ip +
      ":" +
      item.port;


    if (seen.has(key)) {
      continue;
    }


    seen.add(key);

    uniqueItems.push(item);
  }


  // ----------------------------------------------------------
  // 6. 生成 VLESS
  // ----------------------------------------------------------

  const nodes = [];


  for (
    let i = 0;
    i < uniqueItems.length;
    i++
  ) {

    const item =
      uniqueItems[i];


    const node =
      makeVLESS(
        item.ip,
        item.port,
        i + 1
      );


    nodes.push(node);
  }


  // ----------------------------------------------------------
  // 7. 最终结果
  // ----------------------------------------------------------

  return {
    nodes:
      nodes,

    output:
      nodes.join("\n")
  };
}


// ============================================================
// 解析 IP:PORT#备注
// ============================================================

function parseLine(line) {

  line =
    line.trim();


  if (!line) {
    return null;
  }


  const hashIndex =
    line.indexOf("#");


  if (hashIndex === -1) {
    return null;
  }


  const address =
    line
      .substring(
        0,
        hashIndex
      )
      .trim();


  const name =
    line
      .substring(
        hashIndex + 1
      )
      .trim();


  const match =
    address.match(
      /^(\d{1,3}(?:\.\d{1,3}){3}):(\d+)$/
    );


  if (!match) {
    return null;
  }


  return {

    ip:
      match[1],

    port:
      match[2],

    name:
      name
  };
}


// ============================================================
// 新加坡识别
// ============================================================

function detectCountry(name) {

  const keywords = [

    "SG",
    "Singapore",
    "新加坡"
  ];


  for (const keyword of keywords) {

    if (keyword === "SG") {

      const reg =
        /(^|[^A-Za-z])SG([^A-Za-z]|$)/i;


      if (reg.test(name)) {
        return "SG";
      }

    } else {

      if (
        name
          .toLowerCase()
          .includes(
            keyword.toLowerCase()
          )
      ) {
        return "SG";
      }
    }
  }


  return null;
}


// ============================================================
// 提取 AS
// ============================================================

function detectAS(name) {

  const match =
    name.match(
      /\bAS\s*(\d+)\b/i
    );


  if (!match) {
    return null;
  }


  return "AS" + match[1];
}


// ============================================================
// IP → 数字
// ============================================================

function ipToNumber(ip) {

  return ip
    .split(".")
    .reduce(
      function(result, part) {

        return (
          result * 256 +
          Number(part)
        );

      },
      0
    );
}


// ============================================================
// VLESS
// ============================================================

function makeVLESS(
  ip,
  port,
  index
) {

  const name =
    "新加坡🇸🇬⚡️-" +
    index;


  return (
    "vless://" +
    UUID +
    "@" +
    ip +
    ":" +
    port +

    "?encryption=none" +

    "&security=tls" +

    "&sni=" +
    DOMAIN +

    "&type=ws" +

    "&host=" +
    DOMAIN +

    "&path=%2Fvless-ws" +

    "#" +
    name
  );
}
